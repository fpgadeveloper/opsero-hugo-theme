{{- /* Markdown images: a plain with the image's size, loaded lazily. The destination is looked up in the page bundle, then in the site's assets (e.g. "images/board/top.webp" for assets/images/board/top.webp). Images are served as they are (no resizing at build time), so optimize them when they are added. */ -}} {{- $u := urls.Parse .Destination -}} {{- $img := "" -}} {{- if not $u.IsAbs }} {{- $path := strings.TrimPrefix "/" $u.Path }} {{- $img = or (.Page.Resources.Get $u.Path) (resources.Get $path) }} {{- end -}} {{- with $img -}} {{- partial "image.html" (dict "src" . "alt" $.PlainText "title" $.Title) -}} {{- else -}} {{- end -}}
